// ESCUTA A TELEMETRIA EM TEMPO REAL DO CELULAR db.ref('telemetria').on('value', (snapshot) => { const data = snapshot.val(); if (data) { // 1. ATUALIZAÇÃO DA BÚSSOLA if (data.bussula !== undefined) { const angulo = data.bussula; // Atualiza o texto em graus (ID correto: numBussula) const txtGraus = document.getElementById('numBussula'); if (txtGraus) txtGraus.innerText = Math.round(angulo); // Rotaciona o SVG da Bússola (ID correto: compassSvg) const bussolaGf = document.getElementById('compassSvg'); if (bussolaGf) { bussolaGf.style.transform = `rotate(${-angulo}deg)`; } } // 2. ATUALIZAÇÃO DO MAPA (GPS) if (data.lat !== undefined && data.lng !== undefined) { const novaLat = data.lat; const novaLng = data.lng; // Atualiza o texto das coordenadas abaixo da bússola const txtGeo = document.getElementById('geoCoord'); if (txtGeo) txtGeo.innerText = `Lat: ${novaLat.toFixed(5)}, Lng: ${novaLng.toFixed(5)}`; // Move o marcador vermelho do robô no mapa para a nova posição roboMarker.setLatLng([novaLat, novaLng]); // Centraliza o mapa suavemente na nova coordenada do robô map.panTo([novaLat, novaLng]); } } });
